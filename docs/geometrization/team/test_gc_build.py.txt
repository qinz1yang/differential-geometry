"""Disposable process/lock fixtures only; never starts Lean or Lake."""
import json
import os
from pathlib import Path
import signal
import subprocess
import sys
import tempfile
import time
import unittest

import gc_build as g


class BuildGateTests(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory(prefix="gc-build-test-")
        self.root = Path(self.temp.name)
        self.state = self.root / "state"
        self.jobs = []

    def tearDown(self):
        for job in self.jobs:
            if job.poll() is None:
                job.terminate()
            try:
                job.communicate(timeout=5)
            except subprocess.TimeoutExpired:
                job.kill()
                job.communicate(timeout=5)
        self.temp.cleanup()

    def wrapper(self, mode, command, wait=True):
        code = ("import sys; from pathlib import Path; "
                "sys.path.insert(0, %r); import gc_build as g; "
                "sys.exit(g.run(%r, 'fixture', %r, state=Path(%r), "
                "detector=lambda table, owners: []))" %
                (str(Path(g.__file__).parent), mode, command, str(self.state)))
        job = subprocess.Popen([sys.executable, "-c", code], stdout=subprocess.PIPE,
                               stderr=subprocess.PIPE, text=True)
        self.jobs.append(job)
        if wait:
            out, err = job.communicate(timeout=8)
            return job.returncode, out, err
        return job

    def hold(self, mode="small", seconds=20, extra=""):
        ready = self.root / ("ready-%d" % len(self.jobs))
        command = [sys.executable, "-c",
                   "import os,time,signal,sys; from pathlib import Path; " + extra +
                   "Path(%r).write_text(str(os.getpid())); time.sleep(%s)" %
                   (str(ready), seconds)]
        job = self.wrapper(mode, command, wait=False)
        deadline = time.monotonic() + 8
        while not ready.exists() and job.poll() is None and time.monotonic() < deadline:
            time.sleep(0.03)
        self.assertTrue(ready.exists(), "Fixture child did not start")
        return job, int(ready.read_text())

    def test_two_small_then_busy_and_large_busy(self):
        self.hold()
        self.hold()
        for mode in ("small", "large"):
            code, out, err = self.wrapper(mode, ["/usr/bin/true"])
            self.assertEqual(code, 75)
            self.assertIn("busy", err)

    def test_large_excludes_both_modes(self):
        self.hold("large")
        for mode in ("small", "large"):
            self.assertEqual(self.wrapper(mode, ["/usr/bin/true"])[0], 75)

    def test_exit_output_and_runtime_default(self):
        code, out, err = self.wrapper("small", [sys.executable, "-c",
            "import os,sys; print('OUT',os.environ['LEAN_NUM_THREADS']); "
            "print('ERR',file=sys.stderr); sys.exit(23)"])
        self.assertEqual(code, 23)
        self.assertEqual(out, "OUT " + os.environ.get("LEAN_NUM_THREADS", "4") + "\n")
        self.assertEqual(err, "ERR\n")
        self.assertEqual(self.wrapper("large", ["/usr/bin/true"])[0], 0)

    def test_term_forwarded_to_own_child(self):
        other, _ = self.hold()
        job, _ = self.hold(extra="signal.signal(signal.SIGTERM, lambda a,b: sys.exit(43)); ")
        job.terminate()
        job.communicate(timeout=5)
        self.assertEqual(job.returncode, 43)
        self.assertIsNone(other.poll())
        other.terminate()
        other.communicate(timeout=5)
        self.assertEqual(self.wrapper("large", ["/usr/bin/true"])[0], 0)

    def test_child_keeps_locks_when_wrapper_killed(self):
        job, childpid = self.hold("large", seconds=2)
        job.kill()
        job.wait(timeout=5)
        self.assertEqual(self.wrapper("small", ["/usr/bin/true"])[0], 75)
        owners, stale = g.owner_records(self.state, g.process_table())
        self.assertEqual(len(owners), 1)
        self.assertEqual(owners[0]["child"]["pid"], childpid)
        time.sleep(2.2)
        self.assertEqual(self.wrapper("large", ["/usr/bin/true"])[0], 0)

    def test_stale_metadata_cannot_hold_capacity(self):
        self.state.mkdir()
        (self.state / "owner-stale.json").write_text(json.dumps({
            "wrapper": {"pid": os.getpid(), "start": "wrong identity"},
            "child": None}))
        self.assertEqual(g.owner_records(self.state, g.process_table())[0], [])
        self.assertEqual(self.wrapper("large", ["/usr/bin/true"])[0], 0)

    def test_process_classification_and_managed_descendants(self):
        table = {100: {"pid": 100, "ppid": 1, "start": "one"},
                 101: {"pid": 101, "ppid": 100, "start": "two"},
                 102: {"pid": 102, "ppid": 101, "start": "three"},
                 200: {"pid": 200, "ppid": 1, "start": "four"}}
        details = lambda pid: ("/tools/lean", ["lean", "File.lean"])
        cwd = lambda pid: str(g.DEVELOPER / "gc-fixture")
        owners = [{"wrapper": {"pid": 100, "start": "one"}, "child": None}]
        self.assertEqual([p["pid"] for p in g.unwrapped_builds(table, owners, details, cwd)], [200])
        self.assertFalse(g.is_build("/bin/sh", ["sh", "-c", "lean fake.lean"]))
        self.assertFalse(g.is_build("/bin/python3", ["python3", "-c", "check_integration.py"]))
        self.assertFalse(g.is_build("/tools/lean", ["lean", "--server"]))
        self.assertFalse(g.is_build("/tools/lean", ["lean", "--help"]))
        self.assertFalse(g.is_build("/tools/lake", ["lake", "env", "lean", "--server"]))
        self.assertTrue(g.is_build("/tools/lake", ["lake", "build", "DifferentialGeometry"]))
        self.assertTrue(g.is_build("/tools/lake", ["lake", "-v", "build", "DifferentialGeometry"]))
        self.assertTrue(g.is_build("/bin/Python", ["Python", "tools/gc/check_integration.py"]))
        self.assertFalse(g.is_build("/bin/Python", ["Python", "tools/gc/check_integration.py", "--static"]))


if __name__ == "__main__":
    unittest.main(verbosity=2)
