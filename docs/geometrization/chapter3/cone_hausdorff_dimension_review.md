private theorem countable_nonempty_base_exact_one (Y : Type*) [MetricSpace Y]
    [Countable Y] [Nonempty Y] : dimH (univ : Set (EuclideanCone Y)) = 1 := by
  have hbase : dimH (univ : Set Y) = 0 := (to_countable _).dimH_zero
  have hupper := dimH_univ_le_dimH_base_add_one (Y := Y)
  rw [hbase, zero_add] at hupper
  have hI : dimH (Icc (0 : ℝ) 1) = 1 := by
    have hi : (interior (Icc (0 : ℝ) 1)).Nonempty := by
      rw [interior_Icc]
      exact ⟨1 / 2, by norm_num⟩
    simpa using Real.dimH_of_nonempty_interior hi
  have hsub : dimH (univ : Set (Icc (0 : ℝ) 1)) = 1 := by
    have hi := (isometry_subtype_coe (s := Icc (0 : ℝ) 1)).dimH_image univ
    simpa only [image_univ, Subtype.range_coe, hI] using hi.symm
  let f : Icc (0 : ℝ) 1 → ℝ≥0 := fun t => NNReal.mk t.val t.property.1
  have hf : Isometry f := by
    apply Isometry.of_dist_eq
    intro s t
    rfl
  obtain ⟨u⟩ := ‹Nonempty Y›
  have hi := (isometry_mk u).comp hf
  have himage := hi.dimH_image univ
  have hlower := (dimH_mono (subset_univ (((fun r : ℝ≥0 => mk r u) ∘ f) '' univ)))
  rw [himage, hsub] at hlower
  exact le_antisymm hupper hlower

theorem empty_base_cone_is_zero_and_upper_bound_is_strict :
    dimH (univ : Set (EuclideanCone Empty)) = 0 ∧
      dimH (univ : Set (EuclideanCone Empty)) < dimH (univ : Set Empty) + 1 := by
  have hsub : (univ : Set (EuclideanCone Empty)).Subsingleton := by
    intro x _ y _
    rcases eq_tip_or_eq_mk x with rfl | ⟨r, u, _, _⟩
    · rcases eq_tip_or_eq_mk y with rfl | ⟨s, v, _, _⟩
      · rfl
      · exact v.elim
    · exact u.elim
  have hzero := hsub.dimH_zero
  refine ⟨hzero, ?_⟩
  rw [hzero, (to_countable (univ : Set Empty)).dimH_zero]
  norm_num

theorem singleton_base_cone_has_exact_dimension_one :
    dimH (univ : Set (EuclideanCone PUnit)) = 1 :=
  countable_nonempty_base_exact_one PUnit

theorem rational_base_cone_has_exact_dimension_one :
    dimH (univ : Set (EuclideanCone ℚ)) = 1 :=
  countable_nonempty_base_exact_one ℚ
