private theorem plane_segments (x y : Plane) :
    ∃ f : unitInterval → Plane, Continuous f ∧ f 0 = x ∧ f 1 = y ∧
      ∀ s t, dist (f s) (f t) = dist x y * dist s t := by

private theorem same_side_zero {κ : ℝ} (hκ : 0 ≤ κ) (x a b : ℝ) (ha : a ≠ x) (hb : b ≠ x)
    (hs : (x ≤ a ∧ x ≤ b) ∨ (a ≤ x ∧ b ≤ x)) :
    comparisonAngleNegCurvature κ (dist x a) (dist x b) (dist a b) = 0 := by
  have heq : dist a b = |dist x a - dist x b| := by
    rcases hs with ⟨ha', hb'⟩ | ⟨ha', hb'⟩
    · rw [Real.dist_eq a b, Real.dist_eq x a, Real.dist_eq x b,
        abs_of_nonpos (sub_nonpos.mpr ha'), abs_of_nonpos (sub_nonpos.mpr hb')]
      have h : -(x - a) - -(x - b) = a - b := by ring
      rw [h]
    · rw [Real.dist_eq a b, Real.dist_eq x a, Real.dist_eq x b,
        abs_of_nonneg (sub_nonneg.mpr ha'), abs_of_nonneg (sub_nonneg.mpr hb')]
      have h : (x - a) - (x - b) = b - a := by ring
      rw [h, abs_sub_comm]
  rw [heq]
  exact comparisonAngleNegCurvature_abs_sub hκ (dist_pos.mpr ha.symm) (dist_pos.mpr hb.symm)

private theorem real_comparison {κ : ℝ} (hκ : 0 ≤ κ) : fourPointComparison κ (univ : Set ℝ) := by
  intro x _ a _ b _ c _ ha hb hc
  have hab := (comparisonAngleNegCurvature_mem_Icc κ (dist x a) (dist x b) (dist a b)).2
  have hbc := (comparisonAngleNegCurvature_mem_Icc κ (dist x b) (dist x c) (dist b c)).2
  have hca := (comparisonAngleNegCurvature_mem_Icc κ (dist x c) (dist x a) (dist c a)).2
  rcases le_total x a with hxa | hax <;> rcases le_total x b with hxb | hbx <;>
    rcases le_total x c with hxc | hcx
  all_goals first
    | have hz := same_side_zero hκ x a b ha hb (Or.inl ⟨hxa, hxb⟩); linarith
    | have hz := same_side_zero hκ x a b ha hb (Or.inr ⟨hax, hbx⟩); linarith
    | have hz := same_side_zero hκ x b c hb hc (Or.inl ⟨hxb, hxc⟩); linarith
    | have hz := same_side_zero hκ x b c hb hc (Or.inr ⟨hbx, hcx⟩); linarith
    | have hz := same_side_zero hκ x c a hc ha (Or.inl ⟨hxc, hxa⟩); linarith
    | have hz := same_side_zero hκ x c a hc ha (Or.inr ⟨hcx, hax⟩); linarith

private theorem finite_cone_nets {Y : Type*} [MetricSpace Y] [Fintype Y]
    {ε : ℝ} (hε : 0 < ε) (hεone : ε ≤ 1) :
    ∃ S : Finset (EuclideanCone Y),
      (S.card : ℝ) ≤ (1 + 6 * Fintype.card Y) * ε ^ (-(1 : ℝ)) ∧
      ∀ x : EuclideanCone Y, dist x tip ≤ 5 → ∃ y ∈ S, dist x y ≤ ε := by
  classical
  let N : ℕ := ⌊5 / ε⌋₊ + 1
  let r (j : ℕ) : ℝ≥0 := ⟨ε * j, mul_nonneg hε.le (Nat.cast_nonneg _)⟩
  let f : ℕ × Y → EuclideanCone Y := fun a => mk (r a.1) a.2
  let S := insert tip (((Finset.range N).product (Finset.univ : Finset Y)).image f)
  have hN : (N : ℝ) * ε ≤ 6 := by
    have hf := Nat.floor_le (by positivity : 0 ≤ 5 / ε)
    have hm := mul_le_mul_of_nonneg_right hf hε.le
    have he : (5 / ε) * ε = 5 := by field_simp
    dsimp only [N]
    push_cast
    nlinarith
  refine ⟨S, ?_, ?_⟩
  · have hcard : S.card ≤ 1 + N * Fintype.card Y := by
      calc
        S.card ≤ (((Finset.range N).product (Finset.univ : Finset Y)).image f).card + 1 :=
          Finset.card_insert_le _ _
        _ ≤ ((Finset.range N).product (Finset.univ : Finset Y)).card + 1 :=
          Nat.add_le_add_right (Finset.card_image_le) 1
        _ = 1 + N * Fintype.card Y := by
          have hh := Finset.card_product (Finset.range N) (Finset.univ : Finset Y)
          simp only [Finset.card_range, Finset.card_univ] at hh
          exact (congrArg (fun t : ℕ => t + 1) hh).trans (Nat.add_comm _ _)
    have hc : (S.card : ℝ) ≤ 1 + (N : ℝ) * Fintype.card Y := by exact_mod_cast hcard
    rw [Real.rpow_neg_one]
    have hcε := mul_le_mul_of_nonneg_right hc hε.le
    have hNc := mul_le_mul_of_nonneg_right hN (Nat.cast_nonneg (α := ℝ) (Fintype.card Y))
    apply (le_div_iff₀ hε).mpr
    nlinarith
  · intro x hx
    rcases eq_tip_or_eq_mk x with rfl | ⟨ρ, u, hρ, rfl⟩
    · exact ⟨tip, Finset.mem_insert_self _ _, by simpa using hε.le⟩
    · have hρfive : (ρ : ℝ) ≤ 5 := by simpa only [dist_tip, radius_mk] using hx
      let j : ℕ := ⌊(ρ : ℝ) / ε⌋₊
      have hjN : j < N := by
        apply Nat.lt_succ_of_le
        exact Nat.floor_le_floor ((div_le_div_iff_of_pos_right hε).mpr hρfive)
      have hjle : (j : ℝ) ≤ (ρ : ℝ) / ε := Nat.floor_le (by positivity)
      have hjgt : (ρ : ℝ) / ε < (j : ℝ) + 1 := Nat.lt_floor_add_one _
      have hradlo : 0 ≤ (ρ : ℝ) - ε * j := by
        have h := (le_div_iff₀ hε).mp hjle
        nlinarith
      have hradhi : (ρ : ℝ) - ε * j < ε := by
        have h := (div_lt_iff₀ hε).mp hjgt
        nlinarith
      refine ⟨mk (r j) u, Finset.mem_insert_of_mem ?_, ?_⟩
      · apply Finset.mem_image.mpr
        exact ⟨(j, u), Finset.mem_product.mpr ⟨Finset.mem_range.mpr hjN, Finset.mem_univ _⟩, rfl⟩
      · rw [(isometry_mk u).dist_eq]
        change |(ρ : ℝ) - ε * j| ≤ ε
        rw [abs_of_nonneg hradlo]
        exact hradhi.le
