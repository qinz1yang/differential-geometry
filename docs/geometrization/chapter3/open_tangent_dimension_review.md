private theorem affine_short_curves {V : Type*} [NormedAddCommGroup V] [NormedSpace ℝ V] :
    ∀ x y : V, ∀ ε : ℝ, 0 < ε →
      ∃ c : unitInterval → V, Continuous c ∧ c 0 = x ∧ c 1 = y ∧
        eVariationOn c univ < ENNReal.ofReal (dist x y + ε) := by
  apply arbitrarily_short_curves_of_metric_segments
  intro x y
  let f : unitInterval → V := fun t => x + (t : ℝ) • (y - x)
  refine ⟨f, by fun_prop, by simp [f], by simp [f], ?_⟩
  intro s t
  rw [dist_eq_norm, show f s - f t = ((s : ℝ) - t) • (y - x) by
    dsimp [f]; rw [sub_smul]; abel, norm_smul, Real.norm_eq_abs,
    ← dist_eq_norm y x, dist_comm y x]
  change |(s : ℝ) - t| * dist x y = dist x y * |(s : ℝ) - t|
  exact mul_comm _ _
